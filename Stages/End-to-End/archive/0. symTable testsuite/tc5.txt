// -------------------- Scope Cleanup Verification -------------------

SymbolTable table;
GenericSymbol* sym;

table.enterScope();
table.insert("a", new Variable());
table.printTable(symFile);

table.enterScope();
table.insert("b", new Variable());
table.printTable(symFile);

table.exitScope();
table.printTable(symFile);

std::string key = "b";
if (table.lookup(key, sym) == 0) std::cout << key << " Found\n";
else std::cout << key << " Not Found\n";  // b Not Found

key = "a";
if (table.lookup(key, sym) == 0) std::cout << key << " Found\n";  // a Found
else std::cout << key << " Not Found\n";

table.exitScope();

if (table.lookup("a", sym) == 0) std::cout << "a Found\n";
else std::cout << "a Not Found\n";  // a Not Found
table.printTable(symFile);

/*
Expected Output:
b Not Found
a Found
a Not Found

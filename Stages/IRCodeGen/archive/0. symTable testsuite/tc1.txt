// -------------------- Basic Insertion and Lookup -------------------

SymbolTable table;
table.enterScope();

table.insert("x", new Variable());

GenericSymbol* sym;
std::string key = "x";
if (table.lookup(key, sym) == 0) std::cout << key << " Found\n";
else std::cout << key << " Not Found\n";  // x Found

table.printTable(symFile);

key = "b"; // Unknown key Look-up
if (table.lookup(key, sym) == 0) std::cout << key << " Found\n";
else std::cout << key << " Not Found\n"; // b Not Found

table.exitScope();

table.printTable(symFile);

key = "x";
if (table.lookup(key, sym) == 0) std::cout << key << " Found\n";
else std::cout << key << " Not Found\n"; // x Not Found

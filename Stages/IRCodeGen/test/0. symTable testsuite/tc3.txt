// -------------------- Multiple Symbol Types -------------------

SymbolTable table;
GenericSymbol* sym;
table.enterScope();

table.insert("var", new Variable());
table.printTable(symFile);
table.insert("func", new Function());
table.printTable(symFile);
table.insert("enumConst", new EnumConstant());
table.printTable(symFile);

for (auto& key : {"var", "func", "enumConst"}) {
    if (table.lookup(key, sym) == 0) std::cout << key << " Found\n";
    else std::cout << key << " Not Found\n";
}
table.printTable(symFile);

table.exitScope();
if (table.lookup("var", sym) == 0) std::cout << "var Found\n";
else std::cout << "var Not Found\n";
table.printTable(symFile);
/*
Expected Output:
var Found
func Found
enumConst Found
var Not Found
*/

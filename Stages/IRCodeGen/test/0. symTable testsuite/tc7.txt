// -------------------- User-Defined Type (Struct) -------------------

SymbolTable table;
GenericSymbol* sym;
table.enterScope();
table.printTable(symFile);

table.insert("MyStruct", new UserDType());
table.printTable(symFile);

std::string key = "MyStruct";
if (table.lookup(key, sym) == 0) std::cout << key << " Found\n";  // MyStruct Found
else std::cout << key << " Not Found\n";

table.exitScope();
table.printTable(symFile);

if (table.lookup(key, sym) == 0) std::cout << key << " Found\n";
else std::cout << key << " Not Found\n";  // MyStruct Not Found

/*
Expected Output:
MyStruct Found
MyStruct Not Found
*/

// -------------------- Deep Nested Scopes -------------------

SymbolTable table;
GenericSymbol* sym;

table.enterScope();
table.insert("x", new Variable());  // Global x
table.printTable(symFile);

for (int i = 0; i < 5; ++i) {
    table.enterScope();
    table.insert("x", new Variable());  // Shadow x
}

table.printTable(symFile);

std::string key = "x";
if (table.lookup(key, sym) == 0) std::cout << key << " Found (in Scope: " << sym->scopeNo << ")\n";  // x Found (5)
else std::cout << key << " Not Found\n";

for (int i = 0; i < 3; ++i) table.exitScope();
if (table.lookup(key, sym) == 0) std::cout << key << " Found (in Scope: " << sym->scopeNo << ")\n";  // x Found (2)
else std::cout << key << " Not Found\n";
table.printTable(symFile);

for (int i = 0; i < 2; ++i) table.exitScope();
if (table.lookup(key, sym) == 0) std::cout << key << " Found (in Scope: " << sym->scopeNo << ")\n";  // x Found (0)
else std::cout << key << " Not Found\n";
table.printTable(symFile);

table.exitScope();
if (table.lookup(key, sym) == 0) std::cout << key << " Found\n";
else std::cout << key << " Not Found\n";  // x Not Found

table.printTable(symFile);
